    <script>
        let listaCortes = [];
        let descuentos = { top: false, bottom: false, left: false, right: false };
        let unidadActual = 'mm';
        const factoresUnidad = { 'mm': 1, 'cm': 10, 'm': 1000 };
        const KERF_BASE_MM = 3; 
        let kerfActual = 3;     
        
        let idPiezaEnEdicion = null; 

        function cambiarUnidad(nuevaUnidad) {
            if (nuevaUnidad === unidadActual) return;
            document.querySelectorAll('.unit-btn').forEach(btn => btn.classList.remove('active'));
            document.getElementById('btn-' + nuevaUnidad).classList.add('active');

            const factorConversion = factoresUnidad[unidadActual] / factoresUnidad[nuevaUnidad];
            ['hojaAlto', 'hojaAncho', 'piezaAlto', 'piezaAncho'].forEach(id => {
                const el = document.getElementById(id);
                if(el.value) { el.value = Number((parseFloat(el.value) * factorConversion).toFixed(4)); }
            });

            const grosorSelect = document.getElementById('grosor');
            Array.from(grosorSelect.options).forEach(opt => {
                const mmBase = parseFloat(opt.getAttribute('data-mm'));
                const valorConvertido = mmBase / factoresUnidad[nuevaUnidad];
                opt.value = valorConvertido;
                opt.text = `${Number(valorConvertido.toFixed(4))} ${nuevaUnidad}`;
            });

            kerfActual = KERF_BASE_MM / factoresUnidad[nuevaUnidad];

            listaCortes.forEach(p => {
                p.altoFinal = Number((p.altoFinal * factorConversion).toFixed(4));
                p.anchoFinal = Number((p.anchoFinal * factorConversion).toFixed(4));
                p.altoOriginal = Number((p.altoOriginal * factorConversion).toFixed(4));
                p.anchoOriginal = Number((p.anchoOriginal * factorConversion).toFixed(4));
            });

            document.getElementById('lblHojaAlto').innerText = `Hoja Alto (${nuevaUnidad})`;
            document.getElementById('lblHojaAncho').innerText = `Hoja Ancho (${nuevaUnidad})`;
            document.getElementById('lblGrosor').innerText = `Grosor (${nuevaUnidad})`;
            document.getElementById('lblPiezaAlto').innerText = `Alto (${nuevaUnidad})`;
            document.getElementById('lblPiezaAncho').innerText = `Ancho (${nuevaUnidad})`;

            unidadActual = nuevaUnidad;
            actualizarTabla();
            calcular();
            document.getElementById('resultadosOptimizacion').style.display = 'none';
        }

        function toggleDiscount(side) {
            descuentos[side] = !descuentos[side];
            calcular();
        }

        function calcular() {
            const grosor = parseFloat(document.getElementById('grosor').value);
            
            const textoDescuento = `- ${Number(grosor.toFixed(4))} ${unidadActual}`;
            ['btnTop', 'btnBottom', 'btnLeft', 'btnRight'].forEach(id => document.getElementById(id).innerText = textoDescuento);

            ['top', 'bottom', 'left', 'right'].forEach(side => {
                const btn = document.getElementById('btn' + side.charAt(0).toUpperCase() + side.slice(1));
                descuentos[side] ? btn.classList.add('active') : btn.classList.remove('active');
            });

            const alto = parseFloat(document.getElementById('piezaAlto').value) || 0;
            const ancho = parseFloat(document.getElementById('piezaAncho').value) || 0;

            let altoF = Number((alto - (descuentos.top ? grosor : 0) - (descuentos.bottom ? grosor : 0)).toFixed(4));
            let anchoF = Number((ancho - (descuentos.left ? grosor : 0) - (descuentos.right ? grosor : 0)).toFixed(4));

            document.getElementById('textoMedida').innerText = `${altoF} x ${anchoF} ${unidadActual}`;

            const visor = document.getElementById('visualizadorPieza');
            visor.style.borderTop = descuentos.top ? '6px solid #ff6b6b' : '3px solid #8b5a2b';
            visor.style.borderBottom = descuentos.bottom ? '6px solid #ff6b6b' : '3px solid #8b5a2b';
            visor.style.borderLeft = descuentos.left ? '6px solid #ff6b6b' : '3px solid #8b5a2b';
            visor.style.borderRight = descuentos.right ? '6px solid #ff6b6b' : '3px solid #8b5a2b';

            let pWidth = 160, pHeight = 160;
            if (alto > 0 && ancho > 0) {
                if (alto > ancho) { pWidth = (ancho / alto) * 160; } 
                else { pHeight = (alto / ancho) * 160; }
            }
            visor.style.width = Math.max(pWidth, 35) + 'px';
            visor.style.height = Math.max(pHeight, 35) + 'px';

            return { altoF, anchoF };
        }

        function agregarPieza() {
            const medidas = calcular();
            const nombre = document.getElementById('nombrePieza').value || 'Pieza';
            const altoInput = parseFloat(document.getElementById('piezaAlto').value) || 0;
            const anchoInput = parseFloat(document.getElementById('piezaAncho').value) || 0;
            
            if(medidas.altoF <= 0 || medidas.anchoF <= 0) { alert("Las medidas no pueden ser 0 o negativas."); return; }

            if (idPiezaEnEdicion !== null) {
                const index = listaCortes.findIndex(p => p.id === idPiezaEnEdicion);
                if (index !== -1) {
                    listaCortes[index] = { id: idPiezaEnEdicion, nombre: nombre, altoFinal: medidas.altoF, anchoFinal: medidas.anchoF, altoOriginal: altoInput, anchoOriginal: anchoInput, descuentosAplicados: { ...descuentos } };
                }
                idPiezaEnEdicion = null;
                document.getElementById('btnAgregarPieza').innerText = "+ Añadir a Lista de Cortes";
                document.getElementById('btnAgregarPieza').className = "success";
            } else {
                listaCortes.push({ id: Date.now(), nombre: nombre, altoFinal: medidas.altoF, anchoFinal: medidas.anchoF, altoOriginal: altoInput, anchoOriginal: anchoInput, descuentosAplicados: { ...descuentos } });
            }

            document.getElementById('nombrePieza').value = '';
            document.getElementById('nombrePieza').focus();
            actualizarTabla();
            
            if (idPiezaEnEdicion === null) {
                const contenedorTabla = document.querySelector('.table-scroll-container');
                contenedorTabla.scrollTop = contenedorTabla.scrollHeight;
            }
        }

        function editarPieza(id) {
            const pieza = listaCortes.find(p => p.id === id);
            if (!pieza) return;
            idPiezaEnEdicion = id;
            document.getElementById('piezaAlto').value = pieza.altoOriginal;
            document.getElementById('piezaAncho').value = pieza.anchoOriginal;
            document.getElementById('nombrePieza').value = pieza.nombre;
            descuentos = { ...pieza.descuentosAplicados };

            const btnAgregar = document.getElementById('btnAgregarPieza');
            btnAgregar.innerText = "Guardar Cambios";
            btnAgregar.className = "warning"; 
            calcular();
            actualizarTabla();
        }

        function eliminarPieza(id) {
            listaCortes = listaCortes.filter(p => p.id !== id);
            if (idPiezaEnEdicion === id) {
                idPiezaEnEdicion = null;
                document.getElementById('btnAgregarPieza').innerText = "+ Añadir a Lista de Cortes";
                document.getElementById('btnAgregarPieza').className = "success";
            }
            actualizarTabla();
        }

        function actualizarTabla() {
            const tbody = document.querySelector('#tablaCortes tbody');
            tbody.innerHTML = '';
            listaCortes.forEach(p => {
                const trStyle = (p.id === idPiezaEnEdicion) ? "background-color: #fff9e6; border-left: 4px solid #ffc107;" : "";
                tbody.innerHTML += `
                    <tr style="${trStyle}">
                        <td><strong>${p.nombre}</strong></td>
                        <td>${p.altoFinal} x ${p.anchoFinal} ${unidadActual}</td>
                        <td>
                            <div class="actions-cell">
                                <button class="btn-action btn-edit" onclick="editarPieza(${p.id})">Editar</button>
                                <button class="btn-action btn-delete" onclick="eliminarPieza(${p.id})">Quitar</button>
                            </div>
                        </td>
                    </tr>
                `;
            });
        }

        /* --- EXPORTAR DIRECTO A PDF CORREGIDO --- */
        function exportarPDF() {
            if (listaCortes.length === 0) {
                alert("La lista está vacía. Añade piezas antes de exportar.");
                return;
            }

            // 1. Añadimos la clase para ocultar menús y botones
            document.body.classList.add('exporting-pdf');
            
            // 2. Usamos setTimeout para darle tiempo al navegador de ocultar los botones
            // ANTES de que html2pdf tome la captura de la pantalla.
            setTimeout(() => {
                const elemento = document.getElementById('app-container');

                const opciones = {
                    margin:       10,
                    filename:     'Despiece_Melamina.pdf',
                    image:        { type: 'jpeg', quality: 0.98 },
                    html2canvas:  { scale: 2, useCORS: true },
                    jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
                };

                // 3. Generamos el PDF. Usamos ".then()" para asegurarnos de que la 
                // descarga haya terminado ANTES de volver a mostrar los botones.
                html2pdf().set(opciones).from(elemento).save().then(() => {
                    document.body.classList.remove('exporting-pdf');
                }).catch(error => {
                    console.error("Error generando el PDF:", error);
                    document.body.classList.remove('exporting-pdf');
                    alert("Ocurrió un error al generar el PDF.");
                });
                
            }, 300); // 300 milisegundos de espera
        }

        /* --- EXPORTAR DIRECTO A EXCEL --- */
        function exportarExcel() {
            if (listaCortes.length === 0) {
                alert("La lista está vacía. Añade piezas antes de exportar.");
                return;
            }

            let tablaHTML = `
                <tr>
                    <th style="background-color: #0056b3; color: white;">Nombre / Referencia</th>
                    <th style="background-color: #0056b3; color: white;">Alto Final (${unidadActual})</th>
                    <th style="background-color: #0056b3; color: white;">Ancho Final (${unidadActual})</th>
                </tr>
            `;

            listaCortes.forEach(p => {
                tablaHTML += `<tr><td>${p.nombre}</td><td>${p.altoFinal}</td><td>${p.anchoFinal}</td></tr>`;
            });

            let templateExcel = `
                <html xmlns:x="urn:schemas-microsoft-com:office:excel">
                    <head><meta charset="utf-8"></head>
                    <body><h3>Lista de Cortes de Melamina</h3><table border="1">${tablaHTML}</table></body>
                </html>
            `;

            let blob = new Blob([templateExcel], { type: 'application/vnd.ms-excel' });
            let url = URL.createObjectURL(blob);
            
            let enlace = document.createElement('a');
            enlace.href = url;
            enlace.download = 'Lista_de_Cortes.xls';
            document.body.appendChild(enlace);
            enlace.click();
            document.body.removeChild(enlace);
        }

        class Packer {
            constructor(w, h) { this.root = { x: 0, y: 0, w: w, h: h }; }
            fit(blocks) {
                let node;
                for (let n = 0; n < blocks.length; n++) {
                    let block = blocks[n];
                    if (node = this.findNode(this.root, block.w, block.h)) {
                        block.fit = this.splitNode(node, block.w, block.h);
                    } else if (node = this.findNode(this.root, block.h, block.w)) {
                        let temp = block.w; block.w = block.h; block.h = temp;
                        block.rotated = true;
                        block.fit = this.splitNode(node, block.w, block.h);
                    }
                }
            }
            findNode(root, w, h) {
                if (root.used) return this.findNode(root.right, w, h) || this.findNode(root.down, w, h);
                else if ((w <= root.w) && (h <= root.h)) return root;
                else return null;
            }
            splitNode(node, w, h) {
                node.used = true;
                node.down  = { x: node.x, y: node.y + h, w: node.w, h: node.h - h };
                node.right = { x: node.x + w, y: node.y, w: node.w - w, h: h };
                return node;
            }
        }

        function optimizarCortes() {
            if(listaCortes.length === 0) { alert("Añade al menos una pieza a la lista."); return; }

            const hojaW = parseFloat(document.getElementById('hojaAncho').value);
            const hojaH = parseFloat(document.getElementById('hojaAlto').value);

            let bloques = listaCortes.map(p => ({
                id: p.id, nombre: p.nombre, w: p.anchoFinal + kerfActual, h: p.altoFinal + kerfActual,
                originalW: p.anchoFinal, originalH: p.altoFinal
            })).sort((a, b) => (b.w * b.h) - (a.w * a.h));

            let hojasUtilizadas = [];
            let bloquesPendientes = [...bloques];

            while(bloquesPendientes.length > 0) {
                let packer = new Packer(hojaW, hojaH);
                packer.fit(bloquesPendientes);
                
                let hojaActual = { bloquesAsignados: [] };
                let bloquesSobrantes = [];

                bloquesPendientes.forEach(b => {
                    if(b.fit) hojaActual.bloquesAsignados.push(b);
                    else bloquesSobrantes.push(b);
                });

                hojasUtilizadas.push(hojaActual);
                bloquesPendientes = bloquesSobrantes;
            }

            dibujarHojas(hojasUtilizadas, hojaW, hojaH);
        }

        function dibujarHojas(hojas, hojaW, hojaH) {
            document.getElementById('resultadosOptimizacion').style.display = 'block';
            document.getElementById('resumenHojas').innerText = `Total de hojas necesarias: ${hojas.length}`;
            
            const contenedor = document.getElementById('contenedorHojas');
            contenedor.innerHTML = '';

            const escala = 800 / hojaH;
            const visualW = hojaW * escala;
            const visualH = hojaH * escala;

            hojas.forEach((hoja, index) => {
                let htmlHoja = `<div class="board-title">Hoja ${index + 1} (${hojaH} x ${hojaW} ${unidadActual})</div>`;
                htmlHoja += `<div class="board-visual" style="width: ${visualW}px; height: ${visualH}px;">`;

                hoja.bloquesAsignados.forEach(b => {
                    const pW = (b.w - kerfActual) * escala;
                    const pH = (b.h - kerfActual) * escala;
                    const pX = b.fit.x * escala;
                    const pY = b.fit.y * escala;
                    const rotadaText = b.rotated ? '<br><em>(Rotada)</em>' : '';
                    
                    htmlHoja += `
                        <div class="cut-piece" style="left: ${pX}px; top: ${pY}px; width: ${pW}px; height: ${pH}px;">
                            <div><strong>${b.nombre}</strong><br>${b.rotated ? b.originalH : b.originalW} x ${b.rotated ? b.originalW : b.originalH} ${unidadActual} ${rotadaText}</div>
                        </div>
                    `;
                });

                htmlHoja += `</div>`;
                contenedor.innerHTML += htmlHoja;
            });
            
            document.getElementById('resultadosOptimizacion').scrollIntoView({ behavior: 'smooth' });
        }
